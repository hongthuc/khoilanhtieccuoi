    // Lọc sản phẩm theo danh mục
    function filterProducts(categoryName) {
      activeCategory = categoryName;
      renderCategoryNavigation();

      const heading = document.getElementById('current-category-heading');
      if (categoryName === 'all') {
        heading.textContent = 'TẤT CẢ SẢN PHẨM & HÌNH ẢNH THỰC TẾ';
        renderProducts(PRODUCTS);
      } else {
        heading.textContent = 'DANH MỤC: ' + categoryName.toUpperCase();
        const filtered = PRODUCTS.filter(p => {
          if (categoryName === 'Thiết Bị Sân Khấu') {
            return p.category === 'Khinh Khí Cầu' || p.category === 'Phụ Kiện Sân Khấu';
          }
          return p.category === categoryName;
        });
        renderProducts(filtered);
      }
    }

    // Tìm kiếm sản phẩm
    function handleSearchInput(inputId) {
      const val = (document.getElementById(inputId).value || '').trim().toLowerCase();
      if (!val) {
        filterProducts('all');
        return;
      }
      const matched = PRODUCTS.filter(p =>
        p.name.toLowerCase().includes(val) ||
        p.category.toLowerCase().includes(val) ||
        p.subCategory.toLowerCase().includes(val)
      );
      document.getElementById('current-category-heading').textContent = 'KẾT QUẢ TÌM KIẾM: "' + val.toUpperCase() + '"';
      renderProducts(matched);
      document.getElementById('danh-muc-san-pham').scrollIntoView({ behavior: 'smooth' });
    }

    // Mở / đóng menu trên điện thoại
    function toggleMobileMenu(show) {
      const drawer = document.getElementById('mobile-drawer');
      if (show) {
        drawer.classList.remove('hidden');
      } else {
        drawer.classList.add('hidden');
      }
    }
    document.getElementById('mobile-menu-btn').addEventListener('click', () => toggleMobileMenu(true));

    // Mở / đóng Modal xem chi tiết sản phẩm
    function openProductModal(id) {
      const item = PRODUCTS.find(p => p.id === id);
      if (!item) return;
      const modal = document.getElementById('product-modal');
      const img = document.getElementById('modal-image');
      img.setAttribute('data-drive-id', item.driveId);
      img.setAttribute('data-fallback-step', '0');
      img.onerror = function() { handleDriveImgError(this); };
      img.src = 'https://lh3.googleusercontent.com/d/' + item.driveId + '=w900';
      img.alt = item.name;

      document.getElementById('modal-title').textContent = item.name;
      document.getElementById('modal-category').textContent = item.category + ' • ' + item.subCategory;
      document.getElementById('modal-desc').textContent = item.desc;

      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }

    function closeProductModal() {
      const modal = document.getElementById('product-modal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }

    // Xử lý Slider 2 Banner tự trượt chuyên nghiệp
    let currentSlide = 0;
    const totalSlides = 2;
    let slideInterval = null;

    function showSlide(index) {
      currentSlide = (index + totalSlides) % totalSlides;
      const slides = document.querySelectorAll('.hero-slide');
      slides.forEach((slide, idx) => {
        if (idx === currentSlide) {
          slide.classList.remove('hidden', 'opacity-0');
          slide.classList.add('block', 'opacity-100');
        } else {
          slide.classList.remove('block', 'opacity-100');
          slide.classList.add('hidden', 'opacity-0');
        }
      });

      const dot0 = document.getElementById('slide-dot-0');
      const dot1 = document.getElementById('slide-dot-1');
      if (currentSlide === 0) {
        dot0.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
        dot1.className = 'w-2 h-2 rounded-full bg-white/50 transition-all';
      } else {
        dot0.className = 'w-2 h-2 rounded-full bg-white/50 transition-all';
        dot1.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
      }
    }

    function changeSlide(step) {
      showSlide(currentSlide + step);
      resetSlideTimer();
    }

    function goToSlide(index) {
      showSlide(index);
      resetSlideTimer();
    }

    function resetSlideTimer() {
      if (slideInterval) clearInterval(slideInterval);
      slideInterval = setInterval(() => {
        showSlide(currentSlide + 1);
      }, 5000);
    }
